pipeline {
    agent any
    environment {
        DOCKER_IMAGE_NAME = "joselizagaravito/app-java-texto"
    }
    stages {
        stage('Checkout Code') {
            steps {
                // Clona el repositorio manualmente
                git branch: 'master',
                    url: 'https://github.com/joselizagaravito/app-java-texto.git',
                    credentialsId: 'github-credentials-id' // Configura si es un repositorio privado
            }
        }
        stage('Build Application') {
            steps {
                // Compilamos la aplicación y generamos el archivo JAR en 'target/'
                sh 'mvn clean package'
            }
        }
        stage('List Workspace Files') {
            steps {
                // Verificamos que los archivos esperados estén en el directorio de trabajo
                sh 'ls -la'
            }
        }
        stage('Build Docker Image') {
            steps {
                script {
                    // Construimos la imagen Docker usando el Dockerfile y etiquetándola con el número de compilación
                    docker.build("${DOCKER_IMAGE_NAME}:${env.BUILD_NUMBER}")
                }
            }
        }
        stage('Push Docker Image') {
            steps {
                script {
                    // Iniciamos sesión en Docker Hub y subimos la imagen al repositorio
                    docker.withRegistry('https://index.docker.io/v1/', 'docker-credentials-id') {
                        docker.image("${DOCKER_IMAGE_NAME}:${env.BUILD_NUMBER}").push()
                    }
                }
            }
        }
    }
}
